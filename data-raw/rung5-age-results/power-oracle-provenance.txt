#!/bin/bash
#SBATCH -p shared
#SBATCH -c 16
#SBATCH --mem=16G
#SBATCH --time=1-00:00:00
#SBATCH -o /nobackup/pjjg18/evotempo-pr43c/logs/%x-%A_%a.out
# PR #43 follow-up (#69): high-power start-age oracle, 2 starts x 5
# conditionings, 32 chains x NITER iterations each, power design.
set +e; set +o pipefail
export OMP_NUM_THREADS=1 OPENBLAS_NUM_THREADS=1
module purge; module load r/4.5.1 gcc/14.2
W=/nobackup/pjjg18/evotempo-pr43c
export R_LIBS=$W/lib:/nobackup/pjjg18/evotempo-dev/lib
export EVOTEMPO_RUNG5_OUT=$W/results EVOTEMPO_SHA=589f866+overlay
NITER=${NITER:-60000}
cases=(origin origin origin origin origin root root root root root)
vars=(extant joint survival survival-joint none extant joint survival survival-joint none)
i=$((SLURM_ARRAY_TASK_ID - 1))
cd $W/src
/usr/bin/time -v Rscript data-raw/rung5-age-oracle.R ${cases[$i]} $NITER 32 ${vars[$i]} power \
  > $W/out/oracle-${cases[$i]}-${vars[$i]}.txt 2> $W/out/oracle-${cases[$i]}-${vars[$i]}.err
echo "exit $?" >> $W/out/oracle-${cases[$i]}-${vars[$i]}.txt
